import { chmodSync, mkdirSync, mkdtempSync, rmSync, writeFileSync } from "node:fs";
import { tmpdir } from "node:os";
import { join } from "node:path";
import { describe, expect, it } from "vitest";
import { executeOutbound } from "/home/user/juno/packages/registry/src/index.js";
import { SessionLayer, type SessionLimits, anchoringState, gitAnchoring } from "/home/user/juno/packages/enforcement/src/index.js";
import { ordinaryHost } from "/home/user/juno/packages/enforcement/test/layer.js";

const LIMITS: SessionLimits = { rejection_limit: 3, ttl: 3_600_000, budget: { tokens: 1, currency: 1, wall_clock_ms: 3_600_000, turns: 10 }, metered: ["turns", "wall_clock_ms"], clock: "from_admission" };
const CLASS = { sensitivity: "internal" as const, integrity: "operator" as const };
async function git(args: readonly string[]): Promise<void> {
  const r = await executeOutbound({ file: "/usr/bin/git", args: ["-c", "user.name=o", "-c", "user.email=o@o", "-c", "init.defaultBranch=main", ...args] });
  if (r.exitCode !== 0) throw new Error(r.stderr.toString("utf8"));
}

describe("review 7: the service settings name a git program of the runner's own", () => {
  it("a runner-owned --git that answers for the pinned URL with an empty repository", async () => {
    const dir = mkdtempSync(join(tmpdir(), "r7-fakegit-"));
    try {
      const remote = join(dir, "pinned.git");
      await git(["init", "--bare", "--quiet", remote]);
      const seed = join(dir, "seed");
      await git(["clone", "--quiet", remote, seed]);
      writeFileSync(join(seed, "README.md"), "x\n");
      await git(["-C", seed, "add", "README.md"]);
      await git(["-C", seed, "commit", "--quiet", "-m", "s"]);
      await git(["-C", seed, "push", "--quiet", "origin", "HEAD:main"]);
      const work = join(dir, "work"); mkdirSync(work);
      const state = join(dir, "state");
      mkdirSync(state, { mode: 0o700 });
      const journal = join(state, "journal");
      mkdirSync(join(dir, "root-owned"), { mode: 0o755 });
      const pin = join(dir, "root-owned", "anchors.json");
      writeFileSync(pin, JSON.stringify({ url: remote, branch: "main", host_key_fingerprint: null, known_hosts: null, push_key: null, journals: [{ path: journal, recovery_epoch: "2020-01-01T00:00:00Z" }] }), { mode: 0o644 });
      chmodSync(pin, 0o644);
      const honest = { journalKeyDir: join(state, "journal-key"), anchorClone: join(state, "juno-anchors"), intervalMs: 60_000, git: "/usr/bin/git", pin };
      const open = (config: typeof honest): Promise<SessionLayer> => SessionLayer.open({ service: "hook", journal, workspaces: [work], state: anchoringState(config), anchor: null }, { host: ordinaryHost(), anchoring: (root, p) => gitAnchoring(config, root, p) });
      const first = await open(honest);
      first.admit("session-1", LIMITS, { dataClass: CLASS, actor: "probe", payload: { admitted_at: new Date().toISOString(), task: { grants: ["g"] } } });
      expect(await first.close()).toEqual([]);
      const second = await open(honest);
      expect(second.admitted("session-1")).toBe(true);
      expect(await second.close()).toEqual([]);
      // The runner's account: a git of its own (owned by the runner, 0755), named in the service settings it owns.
      const fake = join(dir, "fake-empty.git");
      await git(["init", "--bare", "--quiet", fake]);
      mkdirSync(join(dir, "runner-owned"), { mode: 0o700 });
      const evil = join(dir, "runner-owned", "git");
      writeFileSync(evil, `#!/bin/sh\nfor a in "$@"; do shift; if [ "$a" = "${remote}" ]; then set -- "$@" "${fake}"; else set -- "$@" "$a"; fi; done\nexec /usr/bin/git "$@"\n`, { mode: 0o755 });
      chmodSync(evil, 0o755);
      // ...and the journal and the clone emptied, at the same paths the pin names.
      rmSync(journal, { recursive: true, force: true });
      rmSync(honest.anchorClone, { recursive: true, force: true });
      let third: SessionLayer | null = null;
      let refused: unknown = null;
      try { third = await open({ ...honest, git: evil }); } catch (e) { refused = e; }
      console.log("third start:", third === null ? `REFUSED ${String(refused)}` : `STARTED AS NEW; admitted(session-1)=${String(third.admitted("session-1"))}`);
      if (third !== null) {
        third.admit("session-1", LIMITS, { dataClass: CLASS, actor: "probe", payload: { admitted_at: new Date().toISOString(), task: { grants: ["g"] } } });
        console.log("session-1 admitted again, with a fresh budget");
        await third.close();
      }
      expect(third).toBeNull();
    } finally {
      rmSync(dir, { recursive: true, force: true });
    }
  }, 60_000);
});
